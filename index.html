# 最简单的上传方法
cp 你的文件.pdf ~/hexo-file-demo/source/uploads/
cd ~/hexo-file-demo
hexo clean && hexo generate && hexo deploy
.

│   ├── files/           # 文件库页面

# 启动开发服务器
hexo server

# 重新生成网站
hexo clean && hexo generate

# 部署到GitHub
hexo deploy
