欢迎来到文件资源博客

这是一个支持文件上传和展示的Hexo博客。您可以在这里上传、管理和分享各种文件。

主要功能

📁 文件管理

  • 支持PDF、Word、Excel、PPT等多种格式
  • 文件分类和标签管理
  • 在线预览和下载

🌐 便捷访问

  • 响应式设计，支持移动设备
  • 快速搜索和筛选功能
  • 文件统计信息展示

🔒 简单易用

  • 拖放上传功能
  • 直观的用户界面
  • 无需复杂的配置

快速开始

  1. 查看文件库

    • 浏览已上传的文件
    • 搜索和筛选需要的文件
    • 在线预览或下载

  2. 上传新文件

    • 支持多种文件格式
    • 最大50MB文件大小
    • 支持拖放上传

使用说明

文件格式支持

  • 文档类: PDF, DOC, DOCX, XLS, XLSX, PPT, PPTX, TXT, MD
  • 图片类: JPG, JPEG, PNG, GIF
  • 压缩包: ZIP, RAR, 7Z

上传限制

  • 单个文件大小不超过50MB
  • 每天最多上传10个文件
  • 请勿上传侵权或非法内容

注意事项

  • 文件上传后需要几分钟处理时间
  • 大型文件建议压缩后上传
  • 定期清理不需要的文件

// 从JSON文件加载文件数据
async function loadFileData() {
try {
const response = await fetch(‘/data/files.json’);
if (!response.ok) {
throw new Error(‘网络响应不正常’);
}
const data = await response.json();
return data.files || [];
} catch (error) {
console.error(‘加载文件数据失败:’, error);
return [];
}
}

// 渲染文件列表
async function renderFileList(category = ‘all’) {
const fileGrid = document.getElementById(‘fileGrid’);
fileGrid.innerHTML = ‘

加载中…
‘;

const files = await loadFileData();
const filteredFiles = category === ‘all’
? files
: files.filter(file => file.category === category);

if (filteredFiles.length === 0) {
fileGrid.innerHTML = ‘

暂无文件
‘;
return;
}

fileGrid.innerHTML = filteredFiles.map(file => <div class="file-card" data-type="${file.type}" data-category="${file.category}"> <div class="file-icon"><i class="${file.icon}"></i></div> <div class="file-name">${file.name}</div> <div class="file-meta"> <span class="file-size">${file.size}</span> <span class="file-date">${new Date(file.modified).toLocaleDateString()}</span> </div> <div class="file-actions"> <a href="${file.path}" class="file-download" download="${file.name}"> <i class="fas fa-download"></i> 下载 </a> <button class="file-preview" data-url="${file.path}"> <i class="fas fa-eye"></i> 预览 </button> </div> </div> ).join(‘’);

// 添加预览按钮事件
document.querySelectorAll(‘.file-preview’).forEach(button => {
button.addEventListener(‘click’, () => {
const url = button.getAttribute(‘data-url’);
window.open(url, ‘_blank’);
});
});

// 更新统计信息
updateStats(files);
}

// 更新统计信息
function updateStats(files) {
const totalSize = files.reduce((sum, file) => {
const size = parseFloat(file.size);
return sum + (isNaN(size) ? 0 : size);
}, 0);

document.getElementById(‘total-files’).textContent = files.length;
document.getElementById(‘total-size’).textContent = totalSize.toFixed(1) + ‘ MB’;

if (files.length > 0) {
const latestDate = files.reduce((latest, file) =>
new Date(file.modified) > new Date(latest) ? file.modified : latest, files[0].modified);
document.getElementById(‘last-updated’).textContent = new Date(latestDate).toLocaleDateString();
}
}

// 初始化
document.addEventListener(‘DOMContentLoaded’, async () => {
// 初始渲染
await renderFileList();

// 分类筛选
document.querySelectorAll(‘.category-btn’).forEach(btn => {
btn.addEventListener(‘click’, async () => {
document.querySelectorAll(‘.category-btn’).forEach(b => b.classList.remove(‘active’));
btn.classList.add(‘active’);
await renderFileList(btn.dataset.category);
});
});

// 搜索功能
document.getElementById(‘search-input’).addEventListener(‘input’, async function() {
const searchTerm = this.value.toLowerCase();
const category = document.querySelector(‘.category-btn.active’).dataset.category;

const files = await loadFileData();
let filteredFiles = category === 'all' ? files : files.filter(file => file.category === category);

if (searchTerm) {
  filteredFiles = filteredFiles.filter(file => 
    file.name.toLowerCase().includes(searchTerm)
  );
}

// 更新文件列表显示
const fileGrid = document.getElementById('fileGrid');
if (filteredFiles.length === 0) {
  fileGrid.innerHTML = '<div class="no-files">没有找到符合条件的文件</div>';
  return;
}

fileGrid.innerHTML = filteredFiles.map(file => `

    <div class="file-name">${file.name}</div>
    <div class="file-meta">
      <span class="file-size">${file.size}</span>
      <span class="file-date">${new Date(file.modified).toLocaleDateString()}</span>
    </div>
    <div class="file-actions">
      <a href="${file.path}" class="file-download" download="${file.name}">
        <i class="fas fa-download"></i> 下载
      </a>
      <button class="file-preview" data-url="${file.path}">
        <i class="fas fa-eye"></i> 预览
      </button>
    </div>
  </div>
`).join('');

// 添加预览按钮事件
document.querySelectorAll('.file-preview').forEach(button => {
  button.addEventListener('click', () => {
    const url = button.getAttribute('data-url');
    window.open(url, '_blank');
  });
});

});
});
